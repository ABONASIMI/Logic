% عنصر دوم
deuxieme([_, X | _], X).

% عنصر یکی مانده به آخر
avantDernier([X, _], X).
avantDernier([_ | Q], X) :- avantDernier(Q, X).

% معکوس کردن بهینه با انباشت‌گر (O(n))
inverse(L, Res) :- inv_acc(L, [], Res).
inv_acc([], Acc, Acc).
inv_acc([T | Q], Acc, Res) :- inv_acc(Q, [T | Acc], Res).
